Generate the c file:
py -m cython -3 stein_core.pyx -o stein_core.c

Compile dll for windows:
# cl /LD /O2 /Tc stein_core.c /I "Your/Include/Path" /link /LIBPATH:"Your/libs/Path" /DEF:stein_core.def
cl /LD /O2 /Tc stein_core.c /I "C:\Users\justn\AppData\Local\Python\pythoncore-3.12-64\Include" /link /LIBPATH:"C:\Users\justn\AppData\Local\Python\pythoncore-3.12-64\libs" /DEF:stein_core.def

Compile .so for linux:
gcc -shared -pthread -fPIC -fwrapv -O3 -Wall -fno-strict-aliasing \
    $(python3-config --includes) \
    stein_core.c -o stein_core.so

COmpile raylib linux:
gcc game/core/raylib_server.c -o game/core/raylib_server -I./raylib-5.0_linux_amd64/include ./raylib-5.0_linux_amd64/lib/libraylib.a -lm -lpthread
